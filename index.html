Konfigurasi Google Sheets
Apps Script – Untuk Tulis Balik ke Sheets
⚠️ Punca utama "Failed to Fetch" — Apps Script perlu dikonfigurasi dengan betul untuk terima request dari luar. Guna kod baru di bawah yang menyelesaikan isu CORS.
  1. Buka Google Sheets anda → klik menu Extensions → Apps Script
  2. Padam semua kod sedia ada dalam editor
  3. Tampal kod baru di bawah, kemudian klik 💾 Save
  4. Klik Deploy → New deployment (atau Manage deployments → Edit jika sudah ada)
  5. Pastikan: Type: Web app | Execute as: Me | Who has access: Anyone
  6. Klik Deploy → Copy URL yang diberikan
  7. Tampal URL di bawah dan klik Simpan & Test
// ═══════════════════════════════════════════════════════ // APPS SCRIPT v2 – Dashboard Kejuruteraan Elektrik // Salin KESELURUHAN kod ini ke Apps Script // ═══════════════════════════════════════════════════════ const SPREADSHEET_ID = '1oPbeDl3kw_0Wo2QMLj_uspYuIi1U8FVgLWzPueZJ4mQ'; // ─── Nama tab dalam Google Sheets ────────────────────── // PENTING: key mesti sama dengan nilai 'tab' yang dihantar dari dashboard const SHEET_NAMES = { dv3: 'DV3', // tab bernama DV3 dalam Sheets ppm: 'PPM', // tab bernama PPM dalam Sheets kontraktor: 'KONTRAKTOR' // tab bernama KONTRAKTOR dalam Sheets }; // ─── CORS headers ────────────────────────────────────── function corsHeaders() { return { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Content-Type': 'application/json' }; } // ─── Entry point ─────────────────────────────────────── function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { try { const p = e.parameter || {}; const action = p.action || ''; const tab = p.tab || ''; let result; switch (action) { case 'ping': result = { status: 'ok', msg: 'Sambungan berjaya!', time: new Date().toISOString() }; break; case 'read': result = readSheet(tab); break; case 'append': result = appendRow(tab, JSON.parse(p.row)); break; case 'update': result = updateRow(tab, parseInt(p.rowIndex), JSON.parse(p.row)); break; case 'delete': result = deleteRow(tab, parseInt(p.rowIndex)); break; default: result = { status: 'error', msg: 'Action tidak diketahui: ' + action }; } return buildResponse(result); } catch (err) { return buildResponse({ status: 'error', msg: err.toString() }); } } function buildResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ─── BACA data dari sheet ────────────────────────────── function readSheet(tab) { const name = SHEET_NAMES[tab]; if (!name) return { status: 'error', msg: 'Tab tidak dikenali: ' + tab }; const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName(name); if (!sheet) return { status: 'error', msg: 'Sheet "' + name + '" tidak wujud' }; const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); if (lastRow < 2) return { status: 'ok', headers: [], data: [] }; const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; const rows = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues(); return { status: 'ok', headers: headers, data: rows, total: rows.length }; } // ─── TAMBAH baris baru ───────────────────────────────── function appendRow(tab, row) { const sheet = getSheet(tab); if (!sheet) return { status: 'error', msg: 'Sheet tidak dijumpai' }; sheet.appendRow(row); return { status: 'ok', msg: 'Baris berjaya ditambah', newRow: sheet.getLastRow() }; } // ─── KEMASKINI baris ─────────────────────────────────── function updateRow(tab, rowIndex, row) { const sheet = getSheet(tab); if (!sheet) return { status: 'error', msg: 'Sheet tidak dijumpai' }; if (rowIndex < 2) return { status: 'error', msg: 'rowIndex tidak sah' }; sheet.getRange(rowIndex, 1, 1, row.length).setValues([row]); return { status: 'ok', msg: 'Baris ' + rowIndex + ' dikemas kini' }; } // ─── PADAM baris ─────────────────────────────────────── function deleteRow(tab, rowIndex) { const sheet = getSheet(tab); if (!sheet) return { status: 'error', msg: 'Sheet tidak dijumpai' }; if (rowIndex < 2) return { status: 'error', msg: 'rowIndex tidak sah' }; sheet.deleteRow(rowIndex); return { status: 'ok', msg: 'Baris ' + rowIndex + ' dipadam' }; } // ─── Helper ──────────────────────────────────────────── function getSheet(tab) { const name = SHEET_NAMES[tab]; if (!name) return null; return SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(name); }
